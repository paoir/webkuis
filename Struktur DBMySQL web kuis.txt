Struktur Database MySQL

Nama database: db_kuis

Tabel 1: participants
Field		Type					Keterangan
id			INT AUTO_INCREMENT		Primary Key
nama		VARCHAR(100)			Nama peserta
npm			VARCHAR(20)				Nomor Induk Mahasiswa
score		INT						Nilai akhir peserta
created_at	TIMESTAMP				Waktu submit

Tabel 2: questions
Field		Type					Keterangan
id			INT AUTO_INCREMENT		Primary Key
question	TEXT					Isi pertanyaan
option_a	VARCHAR(255)			Pilihan A
option_b	VARCHAR(255)			Pilihan B
option_c	VARCHAR(255)			Pilihan C
option_d	VARCHAR(255)			Pilihan D
option_e	VARCHAR(255)			Pilihan E
answer		CHAR(1)					Kunci jawaban (A–E)
